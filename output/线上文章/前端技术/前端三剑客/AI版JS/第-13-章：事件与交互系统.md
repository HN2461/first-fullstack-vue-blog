# 第 13 章：事件与交互系统

事件是Web交互的核心机制，让网页能够响应用户的操作和系统状态变化。本章将深入学习JavaScript事件系统，从基础的事件监听到复杂的事件管理模式，构建真正动态的用户界面。

## 13.1 事件监听（addEventListener）

现代JavaScript事件处理的标准方法。

### 基础事件监听

掌握addEventListener的核心用法：

```javascript
/**
 * 事件监听基础示例
 */
const EventListenerExamples = {
    /**
     * 基础语法和用法
     */
    basicUsage() {
        console.log('=== 基础事件监听 ===');
        
        if (typeof document === 'undefined') return;
        
        // 创建测试按钮
        const button = document.createElement('button');
        button.textContent = '点击我';
        button.id = 'test-button';
        document.body.appendChild(button);
        
        // 1. 基础事件监听
        button.addEventListener('click', function(event) {
            console.log('按钮被点击了！');
            console.log('事件类型:', event.type);
            console.log('目标元素:', event.target.tagName);
        });
        
        // 2. 使用箭头函数
        button.addEventListener('mouseenter', (event) => {
            console.log('鼠标进入按钮');
            event.target.style.backgroundColor = 'lightblue';
        });
        
        button.addEventListener('mouseleave', (event) => {
            console.log('鼠标离开按钮');
            event.target.style.backgroundColor = '';
        });
        
        // 3. 命名函数的好处（可以移除）
        function handleDoubleClick(event) {
            console.log('按钮被双击');
            console.log('双击位置:', event.clientX, event.clientY);
        }
        
        button.addEventListener('dblclick', handleDoubleClick);
        
        // 4. 事件监听选项
        button.addEventListener('contextmenu', function(event) {
            console.log('右键菜单事件');
            // event.preventDefault(); // 阻止默认右键菜单
        }, {
            once: true,     // 只执行一次
            passive: false, // 非被动监听
            capture: false  // 冒泡阶段监听
        });
        
        console.log('基础事件监听已设置');
    },
    
    /**
     * 事件监听管理
     */
    eventManagement() {
        console.log('=== 事件监听管理 ===');
        
        if (typeof document === 'undefined') return;
        
        const input = document.createElement('input');
        input.type = 'text';
        input.placeholder = '输入文本测试';
        document.body.appendChild(input);
        
        // 事件处理函数集合
        const handlers = {
            focus: (event) => {
                console.log('输入框获得焦点');
                event.target.style.borderColor = 'blue';
            },
            
            blur: (event) => {
                console.log('输入框失去焦点');
                event.target.style.borderColor = '';
            },
            
            input: (event) => {
                console.log('输入内容:', event.target.value);
            },
            
            keydown: (event) => {
                if (event.key === 'Enter') {
                    console.log('按下回车键');
                }
                if (event.key === 'Escape') {
                    event.target.value = '';
                    console.log('清空输入内容');
                }
            }
        };
        
        // 批量添加事件监听
        Object.keys(handlers).forEach(eventType => {
            input.addEventListener(eventType, handlers[eventType]);
        });
        
        // 移除特定事件监听
        setTimeout(() => {
            input.removeEventListener('input', handlers.input);
            console.log('移除input事件监听');
        }, 10000);
        
        // 事件监听管理类
        class EventManager {
            constructor() {
                this.listeners = new Map();
            }
            
            add(element, type, handler, options = {}) {
                const key = `${element.tagName}-${type}`;
                
                if (!this.listeners.has(key)) {
                    this.listeners.set(key, []);
                }
                
                this.listeners.get(key).push({ handler, options });
                element.addEventListener(type, handler, options);
            }
            
            remove(element, type, handler) {
                const key = `${element.tagName}-${type}`;
                const listeners = this.listeners.get(key);
                
                if (listeners) {
                    const index = listeners.findIndex(l => l.handler === handler);
                    if (index !== -1) {
                        listeners.splice(index, 1);
                        element.removeEventListener(type, handler);
                    }
                }
            }
            
            removeAll(element, type = null) {
                if (type) {
                    const key = `${element.tagName}-${type}`;
                    const listeners = this.listeners.get(key);
                    
                    if (listeners) {
                        listeners.forEach(({ handler }) => {
                            element.removeEventListener(type, handler);
                        });
                        this.listeners.delete(key);
                    }
                } else {
                    // 移除元素的所有事件监听
                    for (let [key, listeners] of this.listeners) {
                        if (key.startsWith(element.tagName)) {
                            listeners.forEach(({ handler }) => {
                                const eventType = key.split('-')[1];
                                element.removeEventListener(eventType, handler);
                            });
                            this.listeners.delete(key);
                        }
                    }
                }
            }
            
            getListenerCount() {
                let count = 0;
                for (let listeners of this.listeners.values()) {
                    count += listeners.length;
                }
                return count;
            }
        }
        
        const eventManager = new EventManager();
        console.log('事件管理器已创建，监听器数量:', eventManager.getListenerCount());
    },
    
    /**
     * 事件对象详解
     */
    eventObjectDetails() {
        console.log('=== 事件对象详解 ===');
        
        if (typeof document === 'undefined') return;
        
        const div = document.createElement('div');
        div.style.cssText = `
            width: 200px;
            height: 100px;
            background: lightgray;
            border: 1px solid black;
            margin: 10px;
            padding: 10px;
            cursor: pointer;
        `;
        div.textContent = '点击测试区域';
        document.body.appendChild(div);
        
        div.addEventListener('click', function(event) {
            console.log('=== 事件对象属性 ===');
            
            // 基本属性
            console.log('type:', event.type);
            console.log('target:', event.target.tagName);
            console.log('currentTarget:', event.currentTarget.tagName);
            console.log('eventPhase:', event.eventPhase);
            
            // 鼠标位置
            console.log('clientX/Y:', event.clientX, event.clientY);
            console.log('pageX/Y:', event.pageX, event.pageY);
            console.log('screenX/Y:', event.screenX, event.screenY);
            console.log('offsetX/Y:', event.offsetX, event.offsetY);
            
            // 修饰键
            console.log('ctrlKey:', event.ctrlKey);
            console.log('shiftKey:', event.shiftKey);
            console.log('altKey:', event.altKey);
            console.log('metaKey:', event.metaKey);
            
            // 鼠标按钮
            console.log('button:', event.button); // 0=左键, 1=中键, 2=右键
            console.log('buttons:', event.buttons); // 按位表示
            
            // 时间戳
            console.log('timeStamp:', event.timeStamp);
            
            // 事件状态
            console.log('bubbles:', event.bubbles);
            console.log('cancelable:', event.cancelable);
            console.log('defaultPrevented:', event.defaultPrevented);
        });
        
        // 自定义事件
        const customEvent = new CustomEvent('customClick', {
            detail: { 
                message: '这是自定义事件',
                timestamp: Date.now()
            },
            bubbles: true,
            cancelable: true
        });
        
        div.addEventListener('customClick', function(event) {
            console.log('自定义事件触发:', event.detail);
        });
        
        // 触发自定义事件
        setTimeout(() => {
            div.dispatchEvent(customEvent);
        }, 2000);
        
        console.log('事件对象详解设置完成');
    }
};
```

### 事件类型分类

了解各种DOM事件类型：

```javascript
/**
 * 事件类型分类示例
 */
const EventTypesExamples = {
    /**
     * UI事件
     */
    uiEvents() {
        console.log('=== UI事件 ===');
        
        if (typeof document === 'undefined') return;
        
        // 窗口事件
        window.addEventListener('load', () => {
            console.log('页面完全加载完成');
        });
        
        window.addEventListener('beforeunload', (event) => {
            console.log('页面即将卸载');
            // event.returnValue = '确定要离开吗？';
        });
        
        window.addEventListener('resize', (event) => {
            console.log('窗口大小改变:', window.innerWidth, 'x', window.innerHeight);
        });
        
        window.addEventListener('scroll', (event) => {
            console.log('页面滚动:', window.pageXOffset, window.pageYOffset);
        });
        
        // 文档事件
        document.addEventListener('DOMContentLoaded', () => {
            console.log('DOM内容加载完成');
        });
        
        document.addEventListener('readystatechange', () => {
            console.log('文档状态变化:', document.readyState);
        });
        
        console.log('UI事件监听已设置');
    },
    
    /**
     * 焦点事件
     */
    focusEvents() {
        console.log('=== 焦点事件 ===');
        
        if (typeof document === 'undefined') return;
        
        const inputs = [];
        for (let i = 0; i < 3; i++) {
            const input = document.createElement('input');
            input.type = 'text';
            input.placeholder = `输入框 ${i + 1}`;
            input.style.margin = '5px';
            
            // focus/blur 不冒泡
            input.addEventListener('focus', (event) => {
                console.log(`输入框 ${i + 1} 获得焦点`);
                event.target.style.backgroundColor = 'lightyellow';
            });
            
            input.addEventListener('blur', (event) => {
                console.log(`输入框 ${i + 1} 失去焦点`);
                event.target.style.backgroundColor = '';
            });
            
            // focusin/focusout 冒泡
            input.addEventListener('focusin', (event) => {
                console.log(`焦点进入 ${i + 1} (冒泡)`);
            });
            
            input.addEventListener('focusout', (event) => {
                console.log(`焦点离开 ${i + 1} (冒泡)`);
            });
            
            document.body.appendChild(input);
            inputs.push(input);
        }
        
        // 焦点管理
        const focusManager = {
            currentIndex: -1,
            
            next() {
                this.currentIndex = (this.currentIndex + 1) % inputs.length;
                inputs[this.currentIndex].focus();
            },
            
            previous() {
                this.currentIndex = this.currentIndex <= 0 ? inputs.length - 1 : this.currentIndex - 1;
                inputs[this.currentIndex].focus();
            }
        };
        
        // 添加导航按钮
        const navDiv = document.createElement('div');
        navDiv.style.margin = '10px';
        
        const prevBtn = document.createElement('button');
        prevBtn.textContent = '上一个';
        prevBtn.addEventListener('click', () => focusManager.previous());
        
        const nextBtn = document.createElement('button');
        nextBtn.textContent = '下一个';
        nextBtn.addEventListener('click', () => focusManager.next());
        
        navDiv.appendChild(prevBtn);
        navDiv.appendChild(nextBtn);
        document.body.appendChild(navDiv);
        
        console.log('焦点事件示例已设置');
    },
    
    /**
     * 设备事件
     */
    deviceEvents() {
        console.log('=== 设备事件 ===');
        
        if (typeof document === 'undefined') return;
        
        // 触摸事件（移动设备）
        const touchArea = document.createElement('div');
        touchArea.style.cssText = `
            width: 200px;
            height: 200px;
            background: lightcoral;
            margin: 10px;
            display: flex;
            align-items: center;
            justify-content: center;
            color: white;
            font-weight: bold;
        `;
        touchArea.textContent = '触摸区域';
        document.body.appendChild(touchArea);
        
        // 触摸事件
        touchArea.addEventListener('touchstart', (event) => {
            console.log('触摸开始，触点数:', event.touches.length);
            event.preventDefault(); // 防止默认行为
        });
        
        touchArea.addEventListener('touchmove', (event) => {
            console.log('触摸移动');
            event.preventDefault();
        });
        
        touchArea.addEventListener('touchend', (event) => {
            console.log('触摸结束');
        });
        
        // 手势事件（Webkit）
        if ('ongesturestart' in touchArea) {
            touchArea.addEventListener('gesturestart', (event) => {
                console.log('手势开始');
            });
            
            touchArea.addEventListener('gesturechange', (event) => {
                console.log('手势变化，缩放:', event.scale, '旋转:', event.rotation);
            });
            
            touchArea.addEventListener('gestureend', (event) => {
                console.log('手势结束');
            });
        }
        
        // 设备方向事件
        if ('onorientationchange' in window) {
            window.addEventListener('orientationchange', (event) => {
                console.log('设备方向改变:', window.orientation);
            });
        }
        
        // 设备运动事件
        if ('ondevicemotion' in window) {
            window.addEventListener('devicemotion', (event) => {
                const acc = event.accelerationIncludingGravity;
                if (acc) {
                    console.log('设备加速度:', acc.x, acc.y, acc.z);
                }
            });
        }
        
        console.log('设备事件监听已设置');
    }
};
```

## 13.2 冒泡与捕获

JavaScript事件流的三个阶段：捕获阶段、目标阶段、冒泡阶段。

### 事件流机制

理解事件在DOM树中的传播过程：

```javascript
/**
 * 事件流机制示例
 */
const EventFlowExamples = {
    /**
     * 冒泡和捕获演示
     */
    bubbleAndCaptureDemo() {
        console.log('=== 冒泡和捕获演示 ===');
        
        if (typeof document === 'undefined') return;
        
        // 创建嵌套结构
        const outer = document.createElement('div');
        const middle = document.createElement('div');
        const inner = document.createElement('div');
        
        outer.style.cssText = `
            width: 300px; height: 300px;
            background: red; padding: 20px;
            border: 2px solid darkred;
        `;
        
        middle.style.cssText = `
            width: 200px; height: 200px;
            background: green; padding: 20px;
            border: 2px solid darkgreen;
        `;
        
        inner.style.cssText = `
            width: 100px; height: 100px;
            background: blue; padding: 20px;
            border: 2px solid darkblue;
            color: white; display: flex;
            align-items: center; justify-content: center;
        `;
        
        outer.id = 'outer';
        middle.id = 'middle';
        inner.id = 'inner';
        inner.textContent = '点击我';
        
        outer.appendChild(middle);
        middle.appendChild(inner);
        document.body.appendChild(outer);
        
        // 捕获阶段监听
        outer.addEventListener('click', (event) => {
            console.log('🔽 outer - 捕获阶段');
        }, true);
        
        middle.addEventListener('click', (event) => {
            console.log('🔽 middle - 捕获阶段');
        }, true);
        
        inner.addEventListener('click', (event) => {
            console.log('🎯 inner - 目标阶段');
        });
        
        // 冒泡阶段监听
        inner.addEventListener('click', (event) => {
            console.log('🔼 inner - 冒泡阶段');
        });
        
        middle.addEventListener('click', (event) => {
            console.log('🔼 middle - 冒泡阶段');
        });
        
        outer.addEventListener('click', (event) => {
            console.log('🔼 outer - 冒泡阶段');
        });
        
        console.log('点击蓝色区域观察事件流');
    },
    
    /**
     * 事件阶段检测
     */
    eventPhaseDetection() {
        console.log('=== 事件阶段检测 ===');
        
        if (typeof document === 'undefined') return;
        
        const container = document.createElement('div');
        const button = document.createElement('button');
        
        container.style.cssText = `
            padding: 20px; background: lightgray;
            margin: 10px; border: 1px solid gray;
        `;
        button.textContent = '测试按钮';
        
        container.appendChild(button);
        document.body.appendChild(container);
        
        // 统一事件处理函数
        function handleEvent(event) {
            const phases = {
                1: 'CAPTURING_PHASE (捕获)',
                2: 'AT_TARGET (目标)',
                3: 'BUBBLING_PHASE (冒泡)'
            };
            
            console.log(`${event.currentTarget.tagName} - ${phases[event.eventPhase]}`);
            console.log('  target:', event.target.tagName);
            console.log('  currentTarget:', event.currentTarget.tagName);
            console.log('  eventPhase:', event.eventPhase);
        }
        
        // 添加各阶段监听
        container.addEventListener('click', handleEvent, true);  // 捕获
        button.addEventListener('click', handleEvent, false);    // 目标/冒泡
        container.addEventListener('click', handleEvent, false); // 冒泡
        
        console.log('事件阶段检测设置完成');
    },
    
    /**
     * 事件路径追踪
     */
    eventPathTracking() {
        console.log('=== 事件路径追踪 ===');
        
        if (typeof document === 'undefined') return;
        
        const section = document.createElement('section');
        const article = document.createElement('article');
        const p = document.createElement('p');
        const span = document.createElement('span');
        
        section.className = 'section';
        article.className = 'article';
        p.className = 'paragraph';
        span.className = 'text';
        span.textContent = '追踪路径';
        
        section.style.cssText = `
            padding: 15px; background: #f0f0f0;
            border: 1px solid #ccc; margin: 10px;
        `;
        
        section.appendChild(article);
        article.appendChild(p);
        p.appendChild(span);
        document.body.appendChild(section);
        
        // 路径追踪处理函数
        function trackEventPath(event) {
            console.log('=== 事件路径信息 ===');
            
            // 使用composedPath()获取完整路径
            if (event.composedPath) {
                const path = event.composedPath();
                console.log('完整事件路径:');
                path.forEach((element, index) => {
                    const name = element.tagName || element.constructor.name;
                    const className = element.className || '';
                    console.log(`  ${index}: ${name} ${className}`);
                });
            }
            
            // 手动追踪路径
            console.log('手动路径追踪:');
            let current = event.target;
            let depth = 0;
            
            while (current) {
                const name = current.tagName || current.constructor.name;
                const className = current.className || '';
                console.log(`  ${depth}: ${name} ${className}`);
                
                current = current.parentElement;
                depth++;
                
                if (depth > 10) break; // 防止无限循环
            }
        }
        
        span.addEventListener('click', trackEventPath);
        
        console.log('事件路径追踪设置完成');
    }
};
```

## 13.3 阻止默认与阻止传播

控制事件的默认行为和传播过程。

### 事件控制方法

掌握preventDefault和stopPropagation的使用：

```javascript
/**
 * 事件控制示例
 */
const EventControlExamples = {
    /**
     * 阻止默认行为
     */
    preventDefaultDemo() {
        console.log('=== 阻止默认行为 ===');
        
        if (typeof document === 'undefined') return;
        
        // 链接示例
        const link = document.createElement('a');
        link.href = 'https://example.com';
        link.textContent = '点击这个链接（已阻止跳转）';
        link.style.display = 'block';
        link.style.margin = '10px';
        
        link.addEventListener('click', (event) => {
            event.preventDefault();
            console.log('链接点击被阻止，不会跳转');
            alert('跳转被阻止！');
        });
        
        document.body.appendChild(link);
        
        // 表单提交示例
        const form = document.createElement('form');
        form.innerHTML = `
            <div style="margin: 10px; padding: 10px; border: 1px solid #ccc;">
                <input type="text" placeholder="输入内容" required>
                <button type="submit">提交（已阻止）</button>
            </div>
        `;
        
        form.addEventListener('submit', (event) => {
            event.preventDefault();
            console.log('表单提交被阻止');
            
            const input = form.querySelector('input');
            if (input.value.trim()) {
                console.log('表单数据:', input.value);
                alert(`数据: ${input.value}`);
            } else {
                alert('请输入内容');
            }
        });
        
        document.body.appendChild(form);
        
        // 右键菜单示例
        const contextDiv = document.createElement('div');
        contextDiv.style.cssText = `
            width: 200px; height: 100px;
            background: lightyellow; border: 1px solid orange;
            display: flex; align-items: center; justify-content: center;
            margin: 10px; cursor: pointer;
        `;
        contextDiv.textContent = '右键点击（菜单被阻止）';
        
        contextDiv.addEventListener('contextmenu', (event) => {
            event.preventDefault();
            console.log('右键菜单被阻止');
            
            // 显示自定义菜单
            const customMenu = document.createElement('div');
            customMenu.style.cssText = `
                position: absolute;
                left: ${event.pageX}px;
                top: ${event.pageY}px;
                background: white;
                border: 1px solid #ccc;
                padding: 5px;
                box-shadow: 2px 2px 5px rgba(0,0,0,0.2);
            `;
            customMenu.innerHTML = `
                <div style="padding: 5px; cursor: pointer;">自定义菜单项1</div>
                <div style="padding: 5px; cursor: pointer;">自定义菜单项2</div>
            `;
            
            document.body.appendChild(customMenu);
            
            // 点击其他地方关闭菜单
            const closeMenu = () => {
                document.body.removeChild(customMenu);
                document.removeEventListener('click', closeMenu);
            };
            setTimeout(() => document.addEventListener('click', closeMenu), 100);
        });
        
        document.body.appendChild(contextDiv);
        
        console.log('阻止默认行为示例设置完成');

                    const control = this.controlledEvents.get(element);

                        console.log(`事件 ${type} 被禁用`);

                        preventDefault: false,

                    });
                }
```

```javascript
                shortcutInfo.textContent = shortcutUsed;
```

```javascript
        this.nextId = 1;
        this.filter = 'all'; // all, active, completed
        
        this.init();
    }
    
    /**
     * 初始化应用
     */
    init() {
        this.render();
        this.bindEvents();
        this.loadFromStorage();
    }
    
    /**
     * 渲染应用结构
     */
    render() {
        this.container.innerHTML = `
            <div class="todo-app">
                <header class="todo-header">
                    <h1>TodoList 事件管理实战</h1>
                    <div class="input-container">
                        <input type="text" 
                               class="todo-input" 
                               placeholder="添加新任务..." 
                               maxlength="100">
                        <button class="add-btn">添加</button>
                    </div>
                </header>
                
                <main class="todo-main">
                    <div class="filter-bar">
                        <button class="filter-btn active" data-filter="all">全部</button>
                        <button class="filter-btn" data-filter="active">未完成</button>
                        <button class="filter-btn" data-filter="completed">已完成</button>
                    </div>
                    
                    <ul class="todo-list"></ul>
                    
                    <div class="todo-stats">
                        <span class="stats-info">总计: <span class="total-count">0</span> 项</span>
                        <button class="clear-completed" style="display: none;">清除已完成</button>
                    </div>
                </main>
            </div>
        `;
        
        // 添加样式
        this.addStyles();
        this.updateDisplay();
    }
    
    /**
     * 绑定所有事件
     */
    bindEvents() {
        // 使用事件委托处理所有交互
        this.container.addEventListener('click', this.handleClick.bind(this));
        this.container.addEventListener('keydown', this.handleKeydown.bind(this));
        this.container.addEventListener('change', this.handleChange.bind(this));
        this.container.addEventListener('input', this.handleInput.bind(this));
        
        // 窗口事件
        window.addEventListener('beforeunload', () => {
            this.saveToStorage();
        });
    }
    
    /**
     * 统一的点击事件处理
     */
    handleClick(event) {
        const target = event.target;
        
        if (target.classList.contains('add-btn')) {
            this.addTodo();
        } else if (target.classList.contains('filter-btn')) {
            this.setFilter(target.dataset.filter);
        } else if (target.classList.contains('clear-completed')) {
            this.clearCompleted();
        } else if (target.classList.contains('delete-btn')) {
            const id = parseInt(target.closest('.todo-item').dataset.id);
            this.deleteTodo(id);
        } else if (target.classList.contains('edit-btn')) {
            const id = parseInt(target.closest('.todo-item').dataset.id);
            this.startEdit(id);
        } else if (target.classList.contains('save-btn')) {
            const id = parseInt(target.closest('.todo-item').dataset.id);
            this.saveEdit(id);
        } else if (target.classList.contains('cancel-btn')) {
            const id = parseInt(target.closest('.todo-item').dataset.id);
            this.cancelEdit(id);
        }
    }
    
    /**
     * 键盘事件处理
     */
    handleKeydown(event) {
        if (event.target.classList.contains('todo-input') && event.key === 'Enter') {
            this.addTodo();
        } else if (event.target.classList.contains('edit-input')) {
            if (event.key === 'Enter') {
                const id = parseInt(event.target.closest('.todo-item').dataset.id);
                this.saveEdit(id);
            } else if (event.key === 'Escape') {
                const id = parseInt(event.target.closest('.todo-item').dataset.id);
                this.cancelEdit(id);
            }
        }
        
        // 全局快捷键
        if (event.ctrlKey || event.metaKey) {
            switch (event.key) {
                case 'a':
                    event.preventDefault();
                    this.selectAll();
                    break;
                case 'd':
                    event.preventDefault();
                    this.clearCompleted();
                    break;
            }
        }
    }
    
    /**
     * 表单变化事件处理
     */
    handleChange(event) {
        if (event.target.classList.contains('todo-checkbox')) {
            const id = parseInt(event.target.closest('.todo-item').dataset.id);
            this.toggleTodo(id);
        }
    }
    
    /**
     * 输入事件处理
     */
    handleInput(event) {
        if (event.target.classList.contains('todo-input')) {
            // 实时验证输入
            const addBtn = this.container.querySelector('.add-btn');
            addBtn.disabled = !event.target.value.trim();
        }
    }
    
    /**
     * 添加新任务
     */
    addTodo() {
        const input = this.container.querySelector('.todo-input');
        const text = input.value.trim();
        
        if (!text) {
            this.showMessage('请输入任务内容', 'error');
            return;
        }
        
        const todo = {
            id: this.nextId++,
            text: text,
            completed: false,
            createdAt: new Date().toISOString()
        };
        
        this.todos.push(todo);
        input.value = '';
        
        this.updateDisplay();
        this.saveToStorage();
        this.showMessage('任务添加成功', 'success');
    }
    
    /**
     * 切换任务完成状态
     */
    toggleTodo(id) {
        const todo = this.todos.find(t => t.id === id);
        if (todo) {
            todo.completed = !todo.completed;
            this.updateDisplay();
            this.saveToStorage();
        }
    }
    
    /**
     * 删除任务
     */
    deleteTodo(id) {
        if (confirm('确定要删除这个任务吗？')) {
            this.todos = this.todos.filter(t => t.id !== id);
            this.updateDisplay();
            this.saveToStorage();
            this.showMessage('任务已删除', 'info');
        }
    }
    
    /**
     * 开始编辑任务
     */
    startEdit(id) {
        const todoItem = this.container.querySelector(`[data-id="${id}"]`);
        const textSpan = todoItem.querySelector('.todo-text');
        const actionBtns = todoItem.querySelector('.action-btns');
        const editBtns = todoItem.querySelector('.edit-btns');
        
        const currentText = textSpan.textContent;
        textSpan.innerHTML = `<input type="text" class="edit-input" value="${currentText}">`;
        
        actionBtns.style.display = 'none';
        editBtns.style.display = 'block';
        
        const editInput = textSpan.querySelector('.edit-input');
        editInput.focus();
        editInput.select();
    }
    
    /**
     * 保存编辑
     */
    saveEdit(id) {
        const todoItem = this.container.querySelector(`[data-id="${id}"]`);
        const editInput = todoItem.querySelector('.edit-input');
        const newText = editInput.value.trim();
        
        if (!newText) {
            this.showMessage('任务内容不能为空', 'error');
            return;
        }
        
        const todo = this.todos.find(t => t.id === id);
        if (todo) {
            todo.text = newText;
            this.updateDisplay();
            this.saveToStorage();
            this.showMessage('任务已更新', 'success');
        }
    }
    
    /**
     * 取消编辑
     */
    cancelEdit(id) {
        this.updateDisplay();
    }
    
    /**
     * 设置过滤器
     */
    setFilter(filter) {
        this.filter = filter;
        
        // 更新按钮状态
        this.container.querySelectorAll('.filter-btn').forEach(btn => {
            btn.classList.remove('active');
        });
        this.container.querySelector(`[data-filter="${filter}"]`).classList.add('active');
        
        this.updateDisplay();
    }
    
    /**
     * 清除已完成任务
     */
    clearCompleted() {
        const completedCount = this.todos.filter(t => t.completed).length;
        if (completedCount === 0) {
            this.showMessage('没有已完成的任务', 'info');
            return;
        }
        
        if (confirm(`确定要删除 ${completedCount} 个已完成的任务吗？`)) {
            this.todos = this.todos.filter(t => !t.completed);
            this.updateDisplay();
            this.saveToStorage();
            this.showMessage(`已删除 ${completedCount} 个任务`, 'info');
        }
    }
    
    /**
     * 全选/取消全选
     */
    selectAll() {
        const allCompleted = this.todos.every(t => t.completed);
        this.todos.forEach(t => t.completed = !allCompleted);
        this.updateDisplay();
        this.saveToStorage();
        this.showMessage(allCompleted ? '取消全选' : '全选完成', 'info');
    }
    
    /**
     * 更新显示
     */
    updateDisplay() {
        this.renderTodoList();
        this.updateStats();
        this.updateClearButton();
    }
    
    /**
     * 渲染任务列表
     */
    renderTodoList() {
        const todoList = this.container.querySelector('.todo-list');
        const filteredTodos = this.getFilteredTodos();
        
        if (filteredTodos.length === 0) {
            todoList.innerHTML = '<li class="empty-message">暂无任务</li>';
            return;
        }
        
        todoList.innerHTML = filteredTodos.map(todo => `
            <li class="todo-item ${todo.completed ? 'completed' : ''}" data-id="${todo.id}">
                <label class="todo-label">
                    <input type="checkbox" class="todo-checkbox" ${todo.completed ? 'checked' : ''}>
                    <span class="todo-text">${this.escapeHtml(todo.text)}</span>
                </label>
                <div class="action-btns">
                    <button class="edit-btn">编辑</button>
                    <button class="delete-btn">删除</button>
                </div>
                <div class="edit-btns" style="display: none;">
                    <button class="save-btn">保存</button>
                    <button class="cancel-btn">取消</button>
                </div>
            </li>
        `).join('');
    }
    
    /**
     * 获取过滤后的任务
     */
    getFilteredTodos() {
        switch (this.filter) {
            case 'active':
                return this.todos.filter(t => !t.completed);
            case 'completed':
                return this.todos.filter(t => t.completed);
            default:
                return this.todos;
        }
    }
    
    /**
     * 更新统计信息
     */
    updateStats() {
        const totalCount = this.todos.length;
        const activeCount = this.todos.filter(t => !t.completed).length;
        const completedCount = totalCount - activeCount;
        
        const totalCountEl = this.container.querySelector('.total-count');
        const statsInfo = this.container.querySelector('.stats-info');
        
        totalCountEl.textContent = totalCount;
        statsInfo.innerHTML = `
            总计: <span class="total-count">${totalCount}</span> 项 | 
            未完成: ${activeCount} | 
            已完成: ${completedCount}
        `;
    }
    
    /**
     * 更新清除按钮
     */
    updateClearButton() {
        const clearBtn = this.container.querySelector('.clear-completed');
        const completedCount = this.todos.filter(t => t.completed).length;
        
        if (completedCount > 0) {
            clearBtn.style.display = 'block';
            clearBtn.textContent = `清除已完成 (${completedCount})`;
        } else {
            clearBtn.style.display = 'none';
        }
    }
    
    /**
     * 显示消息提示
     */
    showMessage(text, type = 'info') {
        const message = document.createElement('div');
        message.className = `message message-${type}`;
        message.textContent = text;
        message.style.cssText = `
            position: fixed; top: 20px; right: 20px; z-index: 1000;
            padding: 10px 15px; border-radius: 4px; color: white;
            background: ${type === 'success' ? '#4CAF50' : type === 'error' ? '#f44336' : '#2196F3'};
            box-shadow: 0 2px 5px rgba(0,0,0,0.2);
        `;
        
        document.body.appendChild(message);
        
        setTimeout(() => {
            message.remove();
        }, 3000);
    }
    
    /**
     * HTML转义
     */
    escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }
    
    /**
     * 保存到本地存储
     */
    saveToStorage() {
        try {
            localStorage.setItem('todoApp', JSON.stringify({
                todos: this.todos,
                nextId: this.nextId,
                filter: this.filter
            }));
        } catch (e) {
            console.error('保存失败:', e);
        }
    }
    
    /**
     * 从本地存储加载
     */
    loadFromStorage() {
        try {
            const data = localStorage.getItem('todoApp');
            if (data) {
                const parsed = JSON.parse(data);
                this.todos = parsed.todos || [];
                this.nextId = parsed.nextId || 1;
                this.filter = parsed.filter || 'all';
                
                this.setFilter(this.filter);
                this.updateDisplay();
            }
        } catch (e) {
            console.error('加载失败:', e);
        }
    }
    
    /**
     * 添加样式
     */
    addStyles() {
        if (document.querySelector('#todo-app-styles')) return;
        
        const style = document.createElement('style');
        style.id = 'todo-app-styles';
        style.textContent = `
            .todo-app {
                max-width: 600px; margin: 20px auto;
                font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif;

            }
            .todo-header {
                background: #4CAF50; color: white; padding: 20px;
            }
            .todo-header h1 { margin: 0 0 15px 0; font-size: 24px; }
```
