# 第二篇　语法基础——构建语言根基

# 第4章　变量与常量

变量是编程语言的基础概念，JavaScript提供了三种声明变量的方式：`var`、`let`和`const`。理解它们的差异和适用场景，是掌握JavaScript的重要一步。

## 4.1 var、let、const 的根本差异

### 变量声明方式概览

JavaScript提供了三种变量声明关键字，每种都有其特定的行为和适用场景：

| 关键字 | 引入版本 | 作用域 | 可重复声明 | 可重新赋值 | 变量提升 | 暂时性死区 |
|--------|----------|--------|------------|------------|----------|------------|
| `var` | ES3 | 函数作用域 | ✅ 是 | ✅ 是 | ✅ 是 | ❌ 否 |
| `let` | ES6 | 块级作用域 | ❌ 否 | ✅ 是 | 🔶 有但不可访问 | ✅ 是 |
| `const` | ES6 | 块级作用域 | ❌ 否 | ❌ 否 | 🔶 有但不可访问 | ✅ 是 |

### var：传统的变量声明

`var`是JavaScript最早的变量声明方式，具有以下特点：

#### 1. 函数作用域
```javascript
function varScopeExample() {
    if (true) {
        var message = 'Hello World';
    }
    
    // var声明的变量在整个函数内都可访问
    console.log(message); // 输出: Hello World
}

varScopeExample();
```

#### 2. 允许重复声明
```javascript
var name = 'Alice';
var name = 'Bob';      // 不会报错，后面的声明会覆盖前面的
console.log(name);     // 输出: Bob

// 在同一作用域内重复声明
var age = 25;
var age = 30;
console.log(age);      // 输出: 30
```

#### 3. 变量提升（Hoisting）
```javascript
console.log(hoistedVar); // 输出: undefined（而不是报错）

var hoistedVar = 'I am hoisted';

console.log(hoistedVar); // 输出: I am hoisted

// 上述代码等同于：
// var hoistedVar; // 声明被提升到顶部
// console.log(hoistedVar); // undefined
// hoistedVar = 'I am hoisted'; // 赋值留在原位
```

#### 4. 循环中的问题
```javascript
// ❌ 经典的var循环问题
for (var i = 0; i < 3; i++) {
    setTimeout(() => {
        console.log('var循环:', i); // 输出: 3, 3, 3
    }, 100);
}

/**
 * 解决var循环问题的方法：使用立即执行函数
 */
for (var j = 0; j < 3; j++) {
    (function(index) {
        setTimeout(() => {
            console.log('IIFE解决:', index); // 输出: 0, 1, 2
        }, 200);
    })(j);
}
```

### let：块级作用域变量

`let`在ES6中引入，解决了`var`的诸多问题：

#### 1. 块级作用域
```javascript
function letScopeExample() {
    if (true) {
        let blockScoped = 'I am block scoped';
        console.log(blockScoped); // 输出: I am block scoped
    }
    
    // console.log(blockScoped); // ReferenceError: blockScoped is not defined
}

// 更多块级作用域示例
{
    let x = 1;
    console.log(x); // 输出: 1
}
// console.log(x); // ReferenceError: x is not defined

for (let i = 0; i < 3; i++) {
    // i只在这个for循环块内有效
    console.log(`循环内: ${i}`);
}
// console.log(i); // ReferenceError: i is not defined
```

#### 2. 不允许重复声明
```javascript
let userName = 'Alice';
// let userName = 'Bob'; // SyntaxError: Identifier 'userName' has already been declared

// 但可以重新赋值
userName = 'Charlie';
console.log(userName); // 输出: Charlie

// 不同块作用域内可以有同名变量
function differentBlocks() {
    let value = 1;
    
    if (true) {
        let value = 2; // 这是一个不同的变量
        console.log('块内:', value); // 输出: 块内: 2
    }
    
    console.log('块外:', value); // 输出: 块外: 1
}

differentBlocks();
```

#### 3. 解决循环问题
```javascript
// ✅ let完美解决循环问题
for (let i = 0; i < 3; i++) {
    setTimeout(() => {
        console.log('let循环:', i); // 输出: 0, 1, 2
    }, 300);
}

/**
 * 创建多个事件处理器的实用示例
 */
const buttons = ['按钮1', '按钮2', '按钮3'];
const container = document.body;

for (let i = 0; i < buttons.length; i++) {
    const button = document.createElement('button');
    button.textContent = buttons[i];
    
    // 每个按钮都能正确记住自己的索引
    button.addEventListener('click', () => {
        console.log(`点击了第 ${i + 1} 个按钮: ${buttons[i]}`);
    });
    
    container.appendChild(button);
}
```

### const：不可变引用常量

`const`用于声明常量，具有`let`的所有特性，但有额外限制：

#### 1. 必须在声明时初始化
```javascript
// ✅ 正确的const声明
const PI = 3.14159;
const APP_NAME = 'My Application';

// ❌ 错误：const声明必须初始化
// const uninitializedConst; // SyntaxError: Missing initializer in const declaration
```

#### 2. 不可重新赋值
```javascript
const GRAVITY = 9.8;
// GRAVITY = 10; // TypeError: Assignment to constant variable

const user = { name: 'Alice', age: 25 };
// user = { name: 'Bob' }; // TypeError: Assignment to constant variable

// ✅ 但可以修改对象的属性
user.age = 26;
user.city = 'New York';
console.log(user); // 输出: { name: 'Alice', age: 26, city: 'New York' }
```

#### 3. 对象和数组的可变性
```javascript
/**
 * const声明的对象内部是可变的
 */
const config = {
    apiUrl: 'https://api.example.com',
    timeout: 5000,
    headers: {
        'Content-Type': 'application/json'
    }
};

// ✅ 可以修改属性
config.timeout = 10000;
config.headers.Authorization = 'Bearer token123';

// ✅ 可以添加新属性
config.retryCount = 3;

console.log(config);

/**
 * const声明的数组也是可变的
 */
const fruits = ['apple', 'banana'];

// ✅ 可以修改数组内容
fruits.push('orange');
fruits[0] = 'grape';
console.log(fruits); // 输出: ['grape', 'banana', 'orange']

// ❌ 但不能重新赋值整个数组
// fruits = ['new', 'array']; // TypeError
```

#### 4. 创建真正不可变的对象
```javascript
/**
 * 使用Object.freeze()创建不可变对象
 */
const immutableConfig = Object.freeze({
    VERSION: '1.0.0',
    MAX_RETRIES: 3,
    ENDPOINTS: Object.freeze({
        API: '/api/v1',
        AUTH: '/auth'
    })
});

// 尝试修改会被忽略（严格模式下会报错）
immutableConfig.VERSION = '2.0.0';
console.log(immutableConfig.VERSION); // 输出: 1.0.0

/**
 * 深度冻结函数
 * @param {Object} obj - 要冻结的对象
 * @returns {Object} 深度冻结后的对象
 */
function deepFreeze(obj) {
    // 获取对象所有属性名
    Object.getOwnPropertyNames(obj).forEach(prop => {
        const value = obj[prop];
        
        // 如果属性值是对象，递归冻结
        if (value && typeof value === 'object') {
            deepFreeze(value);
        }
    });
    
    return Object.freeze(obj);
}

const deeplyImmutable = deepFreeze({
    settings: {
        theme: 'dark',
        language: 'zh-CN'
    },
    features: ['search', 'filter']
});

// 深层属性也无法修改
deeplyImmutable.settings.theme = 'light'; // 在严格模式下会报错
console.log(deeplyImmutable.settings.theme); // 输出: dark
```

### 三者对比总结

#### 实际应用场景对比

```javascript
/**
 * 实际项目中的使用示例
 */

// 1. 配置常量 - 使用const
const CONFIG = {
    API_BASE_URL: 'https://api.myapp.com',
    CACHE_DURATION: 300000, // 5分钟
    MAX_FILE_SIZE: 10 * 1024 * 1024 // 10MB
};

// 2. 函数内的临时变量 - 使用let
function processUserData(users) {
    let totalAge = 0;
    let validUsers = 0;
    
    for (let i = 0; i < users.length; i++) {
        let user = users[i];
        
        if (user.age && user.age > 0) {
            totalAge += user.age;
            validUsers++;
        }
    }
    
    return validUsers > 0 ? totalAge / validUsers : 0;
}

// 3. 遗留代码或特殊需求 - 可能使用var
function legacyFunction() {
    // var在某些旧代码库中仍然存在
    if (typeof someGlobalVar === 'undefined') {
        var someGlobalVar = 'default value';
    }
    
    return someGlobalVar;
}

/**
 * 模块模式中的变量声明
 */
const UserModule = (function() {
    // 私有变量
    let users = [];
    const MAX_USERS = 100;
    
    // 公共接口
    return {
        /**
         * 添加用户
         * @param {Object} user - 用户对象
         * @returns {boolean} 是否添加成功
         */
        addUser(user) {
            if (users.length >= MAX_USERS) {
                console.warn('用户数量已达上限');
                return false;
            }
            
            users.push(user);
            return true;
        },
        
        /**
         * 获取用户总数
         * @returns {number} 用户数量
         */
        getUserCount() {
            return users.length;
        },
        
        /**
         * 清空所有用户
         */
        clearUsers() {
            users = []; // 重新赋值，所以使用let而不是const
        }
    };
})();
```

#### 选择建议决策树

```
变量声明选择
├── 值是否会改变？
│   ├── 否 → const
│   └── 是 ↓
├── 是否需要块级作用域？
│   ├── 是 → let
│   └── 否 ↓
├── 是否在遗留代码中？
│   ├── 是 → 考虑var（但推荐重构为let/const）
│   └── 否 → let

推荐优先级：const > let > var
```

#### 性能考虑

```javascript
/**
 * 性能测试：不同声明方式的影响
 */
function performanceComparison() {
    const iterations = 1000000;
    
    // 测试const
    console.time('const性能测试');
    for (let i = 0; i < iterations; i++) {
        const value = i * 2;
    }
    console.timeEnd('const性能测试');
    
    // 测试let
    console.time('let性能测试');
    for (let i = 0; i < iterations; i++) {
        let value = i * 2;
        value = value + 1; // 重新赋值
    }
    console.timeEnd('let性能测试');
    
    // 测试var
    console.time('var性能测试');
    for (var i = 0; i < iterations; i++) {
        var value = i * 2;
        value = value + 1;
    }
    console.timeEnd('var性能测试');
}

// 在实际项目中，性能差异通常可以忽略
// 更重要的是代码的可读性和可维护性
performanceComparison();
```

## 4.2 作用域：全局、函数、块级

### 作用域的概念

作用域（Scope）决定了变量和函数的可访问性。JavaScript中有三种主要的作用域：

```
JavaScript作用域层次
├── 全局作用域 (Global Scope)
│   ├── 函数作用域 (Function Scope)
│   │   ├── 块级作用域 (Block Scope)
│   │   └── 块级作用域 (Block Scope)
│   └── 函数作用域 (Function Scope)
└── 模块作用域 (Module Scope)
```

### 全局作用域

全局作用域是最外层的作用域，在全局作用域中声明的变量可以在代码的任何地方访问：

#### 1. 浏览器环境中的全局作用域

```javascript
// 全局变量声明
var globalVar = 'I am global with var';
let globalLet = 'I am global with let';  
const globalConst = 'I am global with const';

// 函数声明也是全局的
function globalFunction() {
    return 'I am a global function';
}

// 在浏览器中，var声明的全局变量会成为window对象的属性
console.log(window.globalVar); // 输出: I am global with var
console.log(window.globalLet); // 输出: undefined
console.log(window.globalConst); // 输出: undefined

/**
 * 检查全局变量的函数
 * @param {string} varName - 变量名称
 */
function checkGlobalVariable(varName) {
    if (varName in window) {
        console.log(`${varName} 是window对象的属性`);
    } else {
        console.log(`${varName} 不是window对象的属性`);
    }
}

checkGlobalVariable('globalVar');   // 是window对象的属性
checkGlobalVariable('globalLet');   // 不是window对象的属性
checkGlobalVariable('globalConst'); // 不是window对象的属性
```

#### 2. 意外的全局变量

```javascript
function createAccidentalGlobal() {
    // ❌ 忘记声明变量，创建了意外的全局变量
    accidentalGlobal = 'Oops, I am global!';
    
    // ✅ 正确的局部变量声明
    let properLocal = 'I am properly local';
}

createAccidentalGlobal();

console.log(window.accidentalGlobal); // 输出: Oops, I am global!
// console.log(properLocal); // ReferenceError: properLocal is not defined

/**
 * 使用严格模式避免意外全局变量
 */
function strictModeExample() {
    'use strict';
    
    // 在严格模式下，这会报错
    // strictVar = 'This will throw an error'; // ReferenceError
    
    let properVariable = 'This is correct';
    return properVariable;
}
```

#### 3. 全局命名空间污染问题

```javascript
// ❌ 污染全局命名空间的问题
var name = 'John';
var age = 25;
var email = 'john@example.com';

function updateProfile() {
    // 全局变量可能被意外修改
}

function resetData() {
    // 可能影响其他模块的全局变量
}

// ✅ 使用命名空间模式避免污染
const MyApp = {
    user: {
        name: 'John',
        age: 25,
        email: 'john@example.com'
    },
    
    /**
     * 更新用户资料
     * @param {Object} newData - 新的用户数据
     */
    updateProfile(newData) {
        Object.assign(this.user, newData);
    },
    
    /**
     * 重置用户数据
     */
    resetData() {
        this.user = {
            name: '',
            age: 0,
            email: ''
        };
    }
};

// 使用命名空间
MyApp.updateProfile({ age: 26 });
console.log(MyApp.user.age); // 输出: 26
```

### 函数作用域

每个函数都创建自己的作用域，函数内部声明的变量只在函数内部可访问：

#### 1. 基本函数作用域

```javascript
function functionScopeExample() {
    var functionVar = 'I am function scoped';
    let functionLet = 'I am also function scoped';
    const functionConst = 'I am function scoped too';
    
    console.log(functionVar);   // 可访问
    console.log(functionLet);   // 可访问
    console.log(functionConst); // 可访问
    
    /**
     * 嵌套函数可以访问外层函数的变量
     */
    function innerFunction() {
        console.log('内层函数访问:', functionVar);
        
        // 内层函数的变量
        let innerVariable = 'I am in inner function';
        
        /**
         * 更深层的嵌套
         */
        function deeperFunction() {
            console.log('更深层访问:', innerVariable);
            console.log('更深层访问外层:', functionVar);
        }
        
        deeperFunction();
    }
    
    innerFunction();
    
    // console.log(innerVariable); // ReferenceError: innerVariable is not defined
}

functionScopeExample();

// console.log(functionVar); // ReferenceError: functionVar is not defined
```

#### 2. 参数作用域

```javascript
/**
 * 函数参数也有自己的作用域
 * @param {string} param1 - 第一个参数
 * @param {number} param2 - 第二个参数  
 */
function parameterScope(param1, param2 = 'default value') {
    console.log('参数1:', param1);
    console.log('参数2:', param2);
    
    // 参数可以被重新赋值（不推荐）
    param1 = 'modified';
    console.log('修改后的参数1:', param1);
    
    // 参数只在函数内部有效
    let localVar = param1 + param2;
    return localVar;
}

let result = parameterScope('hello', 'world');
console.log(result); // 输出: modifiedworld

// console.log(param1); // ReferenceError: param1 is not defined
```

#### 3. 立即执行函数表达式（IIFE）

```javascript
/**
 * 使用IIFE创建独立的函数作用域
 */
(function() {
    var privateVar = 'I am private';
    let privateLet = 'I am also private';
    
    console.log('IIFE内部:', privateVar);
    
    // 可以向外部暴露接口
    window.MyModule = {
        /**
         * 获取私有变量的公共方法
         * @returns {string} 私有变量的值
         */
        getPrivateVar() {
            return privateVar;
        },
        
        /**
         * 修改私有变量
         * @param {string} newValue - 新值
         */
        setPrivateVar(newValue) {
            privateVar = newValue;
        }
    };
})();

// 外部无法直接访问私有变量
// console.log(privateVar); // ReferenceError

// 但可以通过暴露的接口访问
console.log(MyModule.getPrivateVar()); // 输出: I am private
MyModule.setPrivateVar('Modified private value');
console.log(MyModule.getPrivateVar()); // 输出: Modified private value
```

### 块级作用域

ES6引入的`let`和`const`支持块级作用域，由`{}`包围的区域形成一个块：

#### 1. 基本块级作用域

```javascript
{
    let blockLet = 'I am block scoped';
    const blockConst = 'I am also block scoped';
    var blockVar = 'I am NOT block scoped'; // var不支持块级作用域
    
    console.log('块内访问let:', blockLet);
    console.log('块内访问const:', blockConst);
    console.log('块内访问var:', blockVar);
}

// console.log(blockLet);   // ReferenceError: blockLet is not defined
// console.log(blockConst); // ReferenceError: blockConst is not defined
console.log(blockVar);      // 输出: I am NOT block scoped

/**
 * if语句的块级作用域
 */
function blockScopeInIf() {
    if (true) {
        let ifLet = 'I am in if block';
        const ifConst = 'I am also in if block';
        var ifVar = 'I am function scoped from if';
        
        console.log('if块内:', ifLet, ifConst, ifVar);
    }
    
    // console.log(ifLet);   // ReferenceError
    // console.log(ifConst); // ReferenceError
    console.log(ifVar);      // 可以访问，因为var是函数作用域
}

blockScopeInIf();
```

#### 2. 循环中的块级作用域

```javascript
/**
 * for循环的块级作用域
 */
function loopBlockScope() {
    console.log('=== let在循环中的表现 ===');
    
    // let在每次循环迭代中都创建新的绑定
    for (let i = 0; i < 3; i++) {
        setTimeout(() => {
            console.log('let循环索引:', i); // 输出: 0, 1, 2
        }, i * 100);
    }
    
    console.log('=== var在循环中的表现 ===');
    
    // var在整个函数作用域中只有一个绑定
    for (var j = 0; j < 3; j++) {
        setTimeout(() => {
            console.log('var循环索引:', j); // 输出: 3, 3, 3
        }, j * 100 + 350);
    }
    
    // console.log(i); // ReferenceError: i is not defined
    console.log('j在循环外:', j); // 输出: 3
}

loopBlockScope();

/**
 * forEach中的作用域
 */
function forEachScope() {
    const items = ['a', 'b', 'c'];
    
    items.forEach((item, index) => {
        // 每个回调函数都有自己的作用域
        let processedItem = `处理后的${item}`;
        
        setTimeout(() => {
            console.log(`索引${index}: ${processedItem}`);
        }, index * 100);
    });
}

forEachScope();
```

#### 3. Switch语句的块级作用域陷阱

```javascript
/**
 * switch语句中的作用域问题
 */
function switchScopeIssue(value) {
    switch (value) {
        case 1:
            let message = 'Case 1'; // 这个let声明在整个switch块中
            console.log(message);
            break;
            
        case 2:
            // let message = 'Case 2'; // SyntaxError: 重复声明
            message = 'Case 2'; // 只能重新赋值
            console.log(message);
            break;
            
        default:
            console.log('Default case');
    }
}

/**
 * 正确的switch块级作用域处理
 */
function switchScopeSolution(value) {
    switch (value) {
        case 1: {
            // 使用花括号创建独立的块级作用域
            let message = 'Case 1';
            console.log(message);
            break;
        }
        
        case 2: {
            // 现在可以重新声明了
            let message = 'Case 2';
            console.log(message);
            break;
        }
        
        default: {
            let message = 'Default case';
            console.log(message);
        }
    }
}

switchScopeSolution(1); // 输出: Case 1
switchScopeSolution(2); // 输出: Case 2
```

### 作用域链

JavaScript引擎通过作用域链来查找变量，从当前作用域开始，向外层作用域查找：

#### 1. 作用域链机制

```javascript
let globalVar = 'Global';

function outerFunction() {
```

## 4.3 变量提升与暂时性死区（TDZ）

### 变量提升机制

变量提升（Hoisting）是JavaScript的一个重要概念，指的是变量和函数声明在编译阶段被移动到其所在作用域的顶部。

#### 提升的本质

```javascript
// 代码编写时的样子
console.log(hoistedVar); // 输出: undefined
var hoistedVar = 'I am hoisted';

// JavaScript引擎实际执行时的理解
var hoistedVar; // 声明被提升
console.log(hoistedVar); // undefined
hoistedVar = 'I am hoisted'; // 赋值留在原位置
```

### var的提升行为

`var`声明会被提升到其作用域的顶部，但只有声明被提升，赋值不会：

```javascript
/**
 * var提升示例
 */
function varHoistingExample() {
    console.log('变量a:', a); // 输出: undefined (而不是报错)
    
    if (true) {
        var a = 'Hello';
        console.log('块内变量a:', a); // 输出: Hello
    }
    
    console.log('块外变量a:', a); // 输出: Hello
    
    // 实际上等同于：
    // var a;
    // console.log('变量a:', a);
    // if (true) {
    //     a = 'Hello';
    //     console.log('块内变量a:', a);
    // }
    // console.log('块外变量a:', a);
}

varHoistingExample();
```

### let和const的暂时性死区

ES6的`let`和`const`也会被提升，但它们有"暂时性死区"（Temporal Dead Zone, TDZ）：

```javascript
/**
 * 暂时性死区示例
 */
function temporalDeadZoneExample() {
    console.log('进入函数作用域');
    
    // 这里是TDZ开始
    // console.log(letVar); // ReferenceError: Cannot access 'letVar' before initialization
    // console.log(constVar); // ReferenceError: Cannot access 'constVar' before initialization
    
    let letVar = 'I am let';     // TDZ结束
    const constVar = 'I am const'; // TDZ结束
    
    console.log(letVar);   // 输出: I am let
    console.log(constVar); // 输出: I am const
}

temporalDeadZoneExample();

/**
 * 复杂的TDZ场景
 */
function complexTDZ() {
    let x = 1;
    
    function innerFunction() {
        // 内部的x声明创建了TDZ，屏蔽了外部的x
        console.log(x); // ReferenceError: Cannot access 'x' before initialization
        let x = 2;
    }
    
    innerFunction();
}

// complexTDZ(); // 取消注释会报错
```

### 函数提升

函数声明会完全提升（包括函数体），而函数表达式不会：

```javascript
/**
 * 函数提升对比
 */
console.log('=== 函数声明提升 ===');
console.log(declaredFunction()); // 输出: I am a declared function

function declaredFunction() {
    return 'I am a declared function';
}

console.log('=== 函数表达式不提升 ===');
// console.log(expressionFunction()); // TypeError: expressionFunction is not a function

var expressionFunction = function() {
    return 'I am a function expression';
};

console.log(expressionFunction()); // 输出: I am a function expression

/**
 * 箭头函数也不会提升
 */
// console.log(arrowFunction()); // ReferenceError: Cannot access 'arrowFunction' before initialization

const arrowFunction = () => {
    return 'I am an arrow function';
};

console.log(arrowFunction()); // 输出: I am an arrow function
```

```javascript
    {
```

### 常见陷阱与避免方法

#### 1. 循环中的变量绑定

```javascript

function createButtonHandlers_Good1() {
    const buttons = document.querySelectorAll('.btn');
    
    for (let i = 0; i < buttons.length; i++) {
        buttons[i].onclick = function() {
            console.log('点击了按钮', i); // 正确输出对应的i值
        };
    }
}

// ✅ 解决方案2：使用forEach
function createButtonHandlers_Good2() {
    const buttons = document.querySelectorAll('.btn');

        button.onclick = () => {

    });
}
```
